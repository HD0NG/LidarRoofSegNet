\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, calc}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily,
  >={Stealth[length=3mm]},
  block/.style={rectangle, rounded corners=3pt, draw=#1, line width=1pt,
                fill=#1!8, text=black, align=center, minimum height=12mm,
                inner sep=4pt, text width=24mm},
  io/.style={block=cInput},
  net/.style={block=cNet},
  contrib/.style={rectangle, rounded corners=3pt, draw=cContrib, line width=1pt,

  out/.style={block=cOut},

% --- main left-to-right chain ---

\node[net, right=8mm of enc] (clu) {HDBSCAN\\clustering};

\node[out, right=8mm of noise] (lab) {Roof-face\\instance labels};

% --- dashed box grouping the refinement, with a contribution banner ---

\node[note, text=cContrib, anchor=south] at (refbox.north) {Plane-aware refinement — contribution (ADR-0001, ADR-0002)};

% --- flow arrows ---
\draw[flow] (pc) -- (enc);
\draw[flow] (enc) -- node[note, above]{$N\times64$} (clu);

\draw[flow] (graph) -- (clf);
\draw[flow] (clf) -- node[note, above]{$P(\text{merge})$} (merge);
\draw[flow] (merge) -- (noise);
\draw[flow] (noise) -- (lab);

\end{tikzpicture}
